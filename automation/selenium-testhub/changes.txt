package com.testhub.tests;

import com.testhub.api.ApiClient;
import com.testhub.api.models.AuthResponse;
import com.testhub.config.ConfigManager;
import com.testhub.constants.FrameworkConstants;
import com.testhub.driver.DriverFactory;
import com.testhub.driver.DriverManager;
import com.testhub.listeners.TestListener;
import com.testhub.pages.DashboardPage;
import com.testhub.reports.ExtentManager;
import com.testhub.pages.LoginPage;
import com.testhub.utils.JavaScriptUtils;
import com.testhub.utils.ScreenshotUtils;
import com.testhub.utils.TestDataFactory;
import org.apache.logging.log4j.LogManager;
import org.apache.logging.log4j.Logger;
import org.testng.annotations.AfterMethod;
import org.testng.annotations.BeforeMethod;
import org.testng.annotations.BeforeSuite;
import org.testng.annotations.Listeners;

import java.io.File;

/**
 * Shared lifecycle for every test:
 * <ul>
 *   <li>warms the cold-start backend once per suite,</li>
 *   <li>spins up a fresh, isolated {@link DriverManager} driver per method
 *       (parallel-safe), and</li>
 *   <li>offers fast, API-based authentication so UI tests start already signed
 *       in instead of re-driving the login form every time.</li>
 * </ul>
 */
@Listeners(TestListener.class)
public abstract class BaseTest {

    protected final Logger log = LogManager.getLogger(getClass());

    @BeforeSuite(alwaysRun = true)
    public void warmUpAndPrepare() {
        for (String dir : new String[]{FrameworkConstants.OUTPUT_DIR, FrameworkConstants.SCREENSHOT_DIR,
                FrameworkConstants.REPORT_DIR, FrameworkConstants.DOWNLOAD_DIR}) {
            new File(dir).mkdirs();
        }
        if (ConfigManager.warmupEnabled()) {
            log.info("Warming up API gateway at {}", ConfigManager.apiBaseUrl());
            ApiClient api = new ApiClient();
            for (int i = 0; i < 8 && !api.isHealthy(); i++) {
                sleep(3000);
            }
            log.info("Gateway healthy: {}", api.isHealthy());

            // The gateway being up doesn't mean core-service is — warm it with a
            // throwaway session so the first dashboard/projects render is fast.
            try {
                ApiClient warm = new ApiClient();
                warm.register(TestDataFactory.fullName(),
                        TestDataFactory.uniqueEmail(), TestDataFactory.password());
                warm.warmDomainServices();
                log.info("Domain services warmed");
            } catch (Exception e) {
                log.warn("Domain warm-up skipped: {}", e.getMessage());
            }
        }
    }

    @BeforeMethod(alwaysRun = true)
    public void setUp() {
        DriverFactory.createDriver();
    }

    @AfterMethod(alwaysRun = true)
    public void tearDown() {
        // Capture screenshot BEFORE quitting — TestNG notifies listeners AFTER @AfterMethod,
        // so by the time onTestFailure/Skipped fires the driver is already gone.
        // Storing it here lets the listener retrieve it via ScreenshotUtils.getStored().
        ScreenshotUtils.captureAndStore();
        DriverManager.quitDriver();
    }

    // --- service health / wake-up -------------------------------------------

    /**
     * Checks all three backend services via their /health endpoints.
     * If any is down, opens the login page and clicks "Services asleep? Wake them",
     * then waits up to 90 s for the success banner before returning.
     * This must be called BEFORE any API call so services are guaranteed up.
     */
    protected void ensureServicesAwake() {
        if (ApiClient.allServicesHealthy()) {
            log.info("All services healthy ✅");
            return;
        }
        log.warn("One or more services are down — waking via UI…");
        new LoginPage().open().wakeServicesAndWait();
        if (ExtentManager.getTest() != null) {
            ExtentManager.getTest().info("ℹ️ Services were asleep — woke them before this login");
        }
    }

    // --- authentication helpers ---------------------------------------------

    /**
     * Logs in entirely over the API, then injects the resulting session into the
     * browser's {@code localStorage} and loads the app already authenticated.
     * This is the default for UI tests that aren't specifically about login.
     */
    protected DashboardPage loginViaApi(String email, String password) {
        ensureServicesAwake();
        ApiClient api = new ApiClient();
        AuthResponse auth = api.login(email, password);
        injectSession(auth);
        return new DashboardPage().waitUntilLoaded();
    }

    /** Registers a brand-new TESTER over the API and signs the browser in as them. */
    protected AuthResponse loginAsNewUser() {
        ensureServicesAwake();
        ApiClient api = new ApiClient();
        AuthResponse auth = api.register(
                TestDataFactory.fullName(), TestDataFactory.uniqueEmail(), TestDataFactory.password());
        ExtentManager.setRunUser(auth.user != null && auth.user.email != null ? auth.user.email : "new-user");
        injectSession(auth);
        new DashboardPage().waitUntilLoaded();
        return auth;
    }

    /**
     * Signs into the browser as a TESTER and returns an authenticated ApiClient
     * ready for data seeding. Checks services are up before making any API call.
     */
    protected ApiClient loginAsTester() {
        ensureServicesAwake();
        ApiClient api = new ApiClient();
        AuthResponse auth;
        if (ConfigManager.reuseUser()) {
            auth = api.login(ConfigManager.testerEmail(), ConfigManager.testerPassword());
            log.info("Reusing tester account: {}", ConfigManager.testerEmail());
            ExtentManager.setRunUser(ConfigManager.testerEmail());
        } else {
            auth = api.register(
                    TestDataFactory.fullName(), TestDataFactory.uniqueEmail(), TestDataFactory.password());
            ExtentManager.setRunUser(auth.user != null && auth.user.email != null ? auth.user.email : "new-user");
        }
        injectSession(auth);
        new DashboardPage().waitUntilLoaded();
        return api;
    }

    /** Logs in (or self-registers) the configured admin and injects the session. */
    protected AuthResponse loginAsAdmin() {
        ensureServicesAwake();
        ApiClient api = new ApiClient();
        AuthResponse auth = api.loginOrRegister("TestHub Admin",
                ConfigManager.adminEmail(), ConfigManager.adminPassword());
        ExtentManager.setRunUser(ConfigManager.adminEmail());
        injectSession(auth);
        new DashboardPage().waitUntilLoaded();
        return auth;
    }

    /** Writes tokens to localStorage on the app origin, then navigates to the dashboard. */
    protected void injectSession(AuthResponse auth) {
        // Navigate to the app origin first so localStorage.setItem is allowed.
        // Services are already awake (ensureServicesAwake was called by the login helper).
        DriverManager.getDriver().get(ConfigManager.baseUrl() + FrameworkConstants.ROUTE_LOGIN);
        JavaScriptUtils.setLocalStorage(FrameworkConstants.LS_ACCESS_TOKEN, auth.accessToken);
        JavaScriptUtils.setLocalStorage(FrameworkConstants.LS_REFRESH_TOKEN, auth.refreshToken);
        DriverManager.getDriver().get(ConfigManager.baseUrl() + FrameworkConstants.ROUTE_DASHBOARD);
    }

    /** Opens the login page for tests that drive the form directly. */
    protected LoginPage openLoginPage() {
        return new LoginPage().open();
    }

    /** Returns an unauthenticated API client for direct API assertions/seeding. */
    protected ApiClient api() {
        return new ApiClient();
    }

    /** Returns an API client already authenticated as a fresh TESTER (for seeding). */
    protected ApiClient apiAsNewUser() {
        ApiClient api = new ApiClient();
        api.register(TestDataFactory.fullName(), TestDataFactory.uniqueEmail(), TestDataFactory.password());
        return api;
    }

    protected void sleep(long millis) {
        try {
            Thread.sleep(millis);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }
}
